-- Draggable Logic
local dragging, dragInput, dragStart, startPos
TopBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

TopBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- ESP Cache Storage
local espCache = {}

local function cleanupESP(player)
    if espCache[player] then
        for _, obj in pairs(espCache[player]) do
            if typeof(obj) == "Instance" then
                obj:Destroy()
            elseif obj.Remove then
                obj:Remove()
            end
        end
        espCache[player] = nil
    end
end

Players.PlayerRemoving:Connect(cleanupESP)

-- Transparent Line-of-Sight Raycast Check
local function hasOpenLOS(origin, targetPos, targetCharacter)
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.IgnoreWater = true
    
    local ignoreList = {LocalPlayer.Character, targetCharacter}
    rayParams.FilterDescendantsInstances = ignoreList
    
    local currentOrigin = origin
    local direction = targetPos - origin
    
    for _ = 1, 8 do
        local result = workspace:Raycast(currentOrigin, direction, rayParams)
        if not result then
            return true -- Completely clear path
        end
        
        local hitPart = result.Instance
        -- Ignore transparent parts or non-collidable parts to shoot through them
        if hitPart.Transparency > 0 or not hitPart.CanCollide then
            table.insert(ignoreList, hitPart)
            rayParams.FilterDescendantsInstances = ignoreList
            currentOrigin = result.Position + (direction.Unit * 0.1)
        else
            return false -- Blocked by a solid opaque wall
        end
    end
    return false
end

-- Main Render Loop
RunService.RenderStepped:Connect(function()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local char = player.Character
            local rootPart = char and char:FindFirstChild("HumanoidRootPart")
            local humanoid = char and char:FindFirstChildOfClass("Humanoid")
            
            if char and rootPart and humanoid and humanoid.Health > 0 then
                if not espCache[player] then
                    espCache[player] = {
                        Beam = Drawing.new("Line"),
                        Box = Drawing.new("Square"),
                        Name = Drawing.new("Text"),
                        Highlight = Instance.new("Highlight")
                    }
                    espCache[player].Beam.Thickness = 1.5
                    espCache[player].Box.Thickness = 1
                    espCache[player].Box.Filled = false
                    espCache[player].Name.Size = 13
                    espCache[player].Name.Center = true
                    espCache[player].Name.Outline = true
                    espCache[player].Name.Color = Color3.fromRGB(255, 255, 255)
                    
                    espCache[player].Highlight.Parent = char
                    espCache[player].Highlight.Enabled = false
                    espCache[player].Highlight.FillColor = Color3.fromRGB(0, 180, 255)
                    espCache[player].Highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                end
                
                local cache = espCache[player]
                local rootPos, onScreen = Camera:WorldToViewportPoint(rootPart.Position)
                
                -- Light Beam Handler
                if getgenv().KitinESP.LightBeam and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                    local myRoot = LocalPlayer.Character.HumanoidRootPart
                    local clearPath = hasOpenLOS(myRoot.Position, rootPart.Position, char)
                    
                    if onScreen and clearPath then
                        local myScreenPos = Camera:WorldToViewportPoint(myRoot.Position)
                        cache.Beam.From = Vector2.new(myScreenPos.X, myScreenPos.Y)
                        cache.Beam.To = Vector2.new(rootPos.X, rootPos.Y)
                        cache.Beam.Color = Color3.fromRGB(0, 255, 100)
                        cache.Beam.Visible = true
                    else
                        cache.Beam.Visible = false
                    end
                else
                    cache.Beam.Visible = false
                end
                
                -- Box ESP Handler
                if getgenv().KitinESP.Box and onScreen then
                    local head = char:FindFirstChild("Head")
                    if head then
                        local headPos = Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
                        local legPos = Camera:WorldToViewportPoint(rootPart.Position - Vector3.new(0, 3, 0))
                        local height = math.abs(headPos.Y - legPos.Y)
                        local width = height / 2
                        
                        cache.Box.Size = Vector2.new(width, height)
                        cache.Box.Position = Vector2.new(rootPos.X - width / 2, headPos.Y)
                        cache.Box.Color = Color3.fromRGB(255, 50, 50)
                        cache.Box.Visible = true
                    else
                        cache.Box.Visible = false
                    end
                else
                    cache.Box.Visible = false
                end
                
                -- Name & Distance Handler
                if (getgenv().KitinESP.NameTag or getgenv().KitinESP.Distance) and onScreen then
                    local labelText = player.Name
                    if getgenv().KitinESP.Distance and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                        local distance = math.floor((LocalPlayer.Character.HumanoidRootPart.Position - rootPart.Position).Magnitude)
                        labelText = labelText .. " [" .. distance .. "m]"
                    end
                    cache.Name.Text = labelText
                    cache.Name.Position = Vector2.new(rootPos.X, rootPos.Y - 45)
                    cache.Name.Visible = true
                else
                    cache.Name.Visible = false
                end
                
                -- Chams Handler
                cache.Highlight.Enabled = getgenv().KitinESP.Chams
            else
                cleanupESP(player)
            end
        end
    end
end)
