    local function createRGBPicker(title, defaultColor, onColorChanged)
        local frame = Instance.new("Frame")
        frame.Size = UDim2.new(0.95, 0, 0, 32)
        frame.BackgroundTransparency = 1
        frame.Parent = SetScrolling
        
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(0.4, 0, 1, 0)
        label.BackgroundTransparency = 1
        label.Text = title
        label.TextColor3 = Color3.fromRGB(255, 255, 255)
        label.Font = Enum.Font.SourceSansBold
        label.TextSize = 11
        label.TextXAlignment = Enum.TextXAlignment.Left
        label.Parent = frame
        
        local function createBox(posX, width, val)
            local box = Instance.new("TextBox")
            box.Size = UDim2.new(width, 0, 0.8, 0)
            box.Position = UDim2.new(posX, 0, 0.1, 0)
            box.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
            box.TextColor3 = Color3.fromRGB(255, 255, 255)
            box.Text = tostring(math.floor(val * 255))
            box.Font = Enum.Font.SourceSansBold
            box.TextSize = 11
            box.ClearTextOnFocus = false
            local cCorner = Instance.new("UICorner")
            cCorner.CornerRadius = UDim.new(0, 4)
            cCorner.Parent = box
            box.Parent = frame
            return box
        end
        
        local rBox = createBox(0.41, 0.18, defaultColor.R)
        local gBox = createBox(0.61, 0.18, defaultColor.G)
        local bBox = createBox(0.81, 0.18, defaultColor.B)
        
        local function update()
            local r = tonumber(rBox.Text)
            local g = tonumber(gBox.Text)
            local b = tonumber(bBox.Text)
            if r and g and b then
                local col = Color3.fromRGB(math.clamp(r, 0, 255), math.clamp(g, 0, 255), math.clamp(b, 0, 255))
                onColorChanged(col)
            end
        end
        
        rBox.FocusLost:Connect(update)
        gBox.FocusLost:Connect(update)
        bBox.FocusLost:Connect(update)
        
        return frame
    end

    -- ==========================================
    -- RUNSERVICE GAME LOOP (V35 PROTECTIONS & ENGINES)
    -- ==========================================
    local vehicleFlyBodyVel, vehicleFlyBodyGyro

    RunService.RenderStepped:Connect(function(dt)
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChild("Humanoid")

        -- 1. Godmode Protection
        if protectionSettings.Godmode and hum then
            hum.Health = hum.MaxHealth
        end

        -- 2. Clean Hazards (Kill Parts, Lava, Traps)
        if protectionSettings.CleanHazards then
            pcall(function()
                for _, obj in pairs(Workspace:GetDescendants()) do
                    if obj:IsA("BasePart") then
                        local nameLower = obj.Name:lower()
                        if nameLower:find("kill") or nameLower:find("hazard") or nameLower:find("lava") or nameLower:find("death") or nameLower:find("damage") or nameLower:find("trap") or nameLower:find("spike") or nameLower:find("acid") then
                            obj.CanTouch = false
                            local touch = obj:FindFirstChild("TouchInterest")
                            if touch then touch:Destroy() end
                        end
                    end
                end
            end)
        end

        -- 3. Anti-NPC Damage
        if protectionSettings.AntiNpcDamage then
            pcall(function()
                for _, model in pairs(Workspace:GetDescendants()) do
                    if model:IsA("Model") and not Players:GetPlayerFromCharacter(model) then
                        local npcHum = model:FindFirstChildOfClass("Humanoid")
                        if npcHum then
                            for _, part in pairs(model:GetDescendants()) do
                                if part:IsA("BasePart") then
                                    local touch = part:FindFirstChild("TouchInterest")
                                    if touch then
                                        touch:Destroy()
                                    end
                                end
                            end
                        end
                    end
                end
            end)
        end

        -- 4. Anti-Fling Protection
        if protectionSettings.AntiFling and hrp then
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        end

        -- 5. Anti-Sit State Lock
        if protectionSettings.AntiSit and hum then
            hum:SetStateEnabled(Enum.HumanoidStateType.Seated, false)
            if hum.Sit then
                hum.Sit = false
            end
        end

        -- 6. Anti-Freeze Protection
        if protectionSettings.AntiFreeze and hrp then
            if hrp.Anchored then
                hrp.Anchored = false
            end
        end

        -- 7. Anti-Crash Safeguard
        if protectionSettings.AntiCrash then
            pcall(function()
                settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
            end)
        end

        -- 8. Anti-Lag Optimizer
        if protectionSettings.AntiLag then
            pcall(function()
                for _, obj in pairs(Workspace:GetDescendants()) do
                    if obj:IsA("ParticleEmitter") or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then
                        obj.Enabled = false
                    elseif obj:IsA("BasePart") then
                        obj.Material = Enum.Material.SmoothPlastic
                        obj.Reflectance = 0
                    end
                end
            end)
        end

        -- 9. TPWalk
        if moveSettings.TPWalkEnabled and hrp and hum then
            local moveDir = hum.MoveDirection
            if moveDir.Magnitude > 0 then
                hrp.CFrame = hrp.CFrame + (moveDir * moveSettings.TPWalkSpeed * dt)
            end
        end

        -- 10. CFrame Walk (CWalk)
        if moveSettings.CWalkEnabled and hrp and hum then
            local moveDir = hum.MoveDirection
            if moveDir.Magnitude > 0 then
                hrp.CFrame = hrp.CFrame + (moveDir * moveSettings.CWalkSpeed * dt)
                hrp.AssemblyLinearVelocity = Vector3.new(0, hrp.AssemblyLinearVelocity.Y, 0)
            end
        end

        -- 11. Vehicle Fly
        if moveSettings.FlyEnabled and hrp and hum then
            local targetPart = hrp
            local isSeated = (hum.SeatPart ~= nil)
            
            if isSeated then
                local seat = hum.SeatPart
                local model = seat:FindFirstAncestorOfClass("Model")
                if model and model:FindFirstChild("PrimaryPart") then
                    targetPart = model.PrimaryPart
                else
                    targetPart = seat
                end
            end

            if not vehicleFlyBodyVel then
                vehicleFlyBodyVel = Instance.new("BodyVelocity")
                vehicleFlyBodyVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                vehicleFlyBodyVel.Velocity = Vector3.new(0, 0, 0)
            end
            vehicleFlyBodyVel.Parent = targetPart

            if not vehicleFlyBodyGyro then
                vehicleFlyBodyGyro = Instance.new("BodyGyro")
                vehicleFlyBodyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
            end
            vehicleFlyBodyGyro.Parent = targetPart
            vehicleFlyBodyGyro.CFrame = camera.CFrame
            
            local camCF = camera.CFrame
            local moveVector = Vector3.new()
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVector = moveVector + camCF.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVector = moveVector - camCF.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVector = moveVector - camCF.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVector = moveVector + camCF.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.E) then moveVector = moveVector + Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.Q) then moveVector = moveVector - Vector3.new(0, 1, 0) end
            
            vehicleFlyBodyVel.Velocity = moveVector * moveSettings.FlySpeed
            
            if not isSeated then
                hum.PlatformStand = true
            end
        else
            if vehicleFlyBodyVel then vehicleFlyBodyVel:Destroy(); vehicleFlyBodyVel = nil end
            if vehicleFlyBodyGyro then vehicleFlyBodyGyro:Destroy(); vehicleFlyBodyGyro = nil end
            if hum and hum.PlatformStand and not moveSettings.CFrameFlyEnabled and not moveSettings.VFlyEnabled then
                hum.PlatformStand = false
                pcall(function() hum:ChangeState(Enum.HumanoidStateType.GettingUp) end)
            end
        end

        -- 12. CFrame Fly (CFly)
        if moveSettings.CFrameFlyEnabled and hrp and hum then
            hum.PlatformStand = true
            local camCF = camera.CFrame
            local moveVector = Vector3.new()
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVector = moveVector + camCF.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVector = moveVector - camCF.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVector = moveVector - camCF.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVector = moveVector + camCF.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.E) then moveVector = moveVector + Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.Q) then moveVector = moveVector - Vector3.new(0, 1, 0) end
            
            if moveVector.Magnitude > 0 then
                hrp.CFrame = hrp.CFrame + (moveVector.Unit * moveSettings.FlySpeed * dt)
            else
                hrp.CFrame = CFrame.new(hrp.Position, hrp.Position + camCF.LookVector)
            end
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        end

        -- 13. Character Fly (VFly)
        if moveSettings.VFlyEnabled and hrp and hum then
            local camCF = camera.CFrame
            local vMove = Vector3.new()
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then vMove = vMove + camCF.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then vMove = vMove - camCF.LookVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then vMove = vMove - camCF.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then vMove = vMove + camCF.RightVector end
            if UserInputService:IsKeyDown(Enum.KeyCode.E) then vMove = vMove + Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.Q) then vMove = vMove - Vector3.new(0, 1, 0) end
            
            hrp.AssemblyLinearVelocity = vMove * moveSettings.FlySpeed
            hrp.CFrame = CFrame.new(hrp.Position, hrp.Position + camCF.LookVector)
            hum.PlatformStand = true
        end

        -- 14. Noclip Engine
        if moveSettings.NoclipEnabled and char then
            pcall(function()
                for _, part in pairs(char:GetDescendants()) do
                    if part:IsA("BasePart") then
                        part.CanCollide = false
                    end
                end
            end)
        end

        -- 15. ESP ENGINE LOOP
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= player then
                local pChar = p.Character
                local pHrp = pChar and pChar:FindFirstChild("HumanoidRootPart")
                local pHum = pChar and pChar:FindFirstChild("Humanoid")
                
                local isTeam = false
                if aimbotSettings.TeamCheck and p.Team and player.Team and p.Team == player.Team then
                    isTeam = true
                end

                local inRange = false
                if hrp and pHrp then
                    local distance = (hrp.Position - pHrp.Position).Magnitude
                    if distance <= espSettings.MaxRange then
                        inRange = true
                    end
                end

                -- Chams
                if espSettings.Enabled and espSettings.Chams and pChar and not isTeam and inRange then
                    if not activeHighlights[p] or activeHighlights[p].Parent ~= pChar then
                        if activeHighlights[p] then activeHighlights[p]:Destroy() end
                        local hl = Instance.new("Highlight")
                        hl.Name = "MulnesChams"
                        hl.Adornee = pChar
                        hl.FillColor = Color3.fromRGB(255, 50, 50)
                        hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                        hl.FillTransparency = 0.5
                        hl.Parent = pChar
                        activeHighlights[p] = hl
                    end
                else
                    if activeHighlights[p] then
                        activeHighlights[p]:Destroy()
                        activeHighlights[p] = nil
                    end
                end

                -- Tracers
                if espSettings.Enabled and espSettings.Tracers and pHrp and not isTeam and inRange then
                    local screenPos, onScreen = camera:WorldToViewportPoint(pHrp.Position)
                    if onScreen then
                        if not activeTracers[p] then
                            local line = Drawing.new("Line")
                            line.Thickness = 1.5
                            line.Color = Color3.fromRGB(255, 100, 100)
                            activeTracers[p] = line
                        end
                        local line = activeTracers[p]
                        line.Visible = true
                        line.From = Vector2.new(camera.ViewportSize.X / 2, camera.ViewportSize.Y)
                        line.To = Vector2.new(screenPos.X, screenPos.Y)
                    else
                        if activeTracers[p] then activeTracers[p].Visible = false end
                    end
                else
                    if activeTracers[p] then
                        activeTracers[p]:Remove()
                        activeTracers[p] = nil
                    end
                end

                -- Sightline View Cone
                if espSettings.Enabled and espSettings.Sightline and pHrp and pHum and not isTeam and inRange then
                    if not activeSightlines[p] then
                        local line = Drawing.new("Line")
                        line.Thickness = 2
                        activeSightlines[p] = line
                    end
                    local line = activeSightlines[p]
                    line.Color = Color3.fromRGB(espSettings.SightlineColorR, espSettings.SightlineColorG, espSettings.SightlineColorB)
                    
                    local head = pChar:FindFirstChild("Head")
                    if head then
                        local headScreen, headOn = camera:WorldToViewportPoint(head.Position)
                        local targetWorldPos = head.Position
                        local dirMode = espSettings.SightlineDirection
                        local length = espSettings.SightlineLength
                        
                        if dirMode == "Front" then
                            targetWorldPos = head.Position + (head.CFrame.LookVector * length)
                        elseif dirMode == "Skyward" then
                            targetWorldPos = head.Position + Vector3.new(0, length, 0)
                        elseif dirMode == "Back" then
                            targetWorldPos = head.Position - (head.CFrame.LookVector * length)
                        elseif dirMode == "Bottom" then
                            targetWorldPos = head.Position - Vector3.new(0, length, 0)
                        end
                        
                        local lookEndScreen, lookOn = camera:WorldToViewportPoint(targetWorldPos)
                        if headOn and lookOn then
                            line.Visible = true
                            line.From = Vector2.new(headScreen.X, headScreen.Y)
                            line.To = Vector2.new(lookEndScreen.X, lookEndScreen.Y)
                        else
                            line.Visible = false
                        end
                    else
                        line.Visible = false
                    end
                else
                    if activeSightlines[p] then
                        activeSightlines[p]:Remove()
                        activeSightlines[p] = nil
                    end
                end

                -- 3D World Light Beam
                if espSettings.Enabled and espSettings.LightBeam and pHrp and not isTeam and inRange then
                    if not activeLightBeams[p] or not activeLightBeams[p].Parent then
                        if activeLightBeams[p] then activeLightBeams[p]:Destroy() end
                        
                        local beamPart = Instance.new("Part")
                        beamPart.Name = "MulnesLightBeamPart"
                        beamPart.Size = Vector3.new(1, 500, 1)
                        beamPart.Anchored = true
                        beamPart.CanCollide = false
                        beamPart.Transparency = 1
                        beamPart.Parent = Workspace
                        
                        local att0 = Instance.new("Attachment", beamPart)
                        att0.Position = Vector3.new(0, -250, 0)
                        local att1 = Instance.new("Attachment", beamPart)
                        att1.Position = Vector3.new(0, 250, 0)
                        
                        local beam = Instance.new("Beam")
                        beam.Name = "MulnesLightBeam"
                        beam.Attachment0 = att0
                        beam.Attachment1 = att1
                        beam.Color = ColorSequence.new(Color3.fromRGB(espSettings.SightlineColorR, espSettings.SightlineColorG, espSettings.SightlineColorB))
                        beam.Width0 = espSettings.BeamWidth
                        beam.Width1 = espSettings.BeamWidth
                        beam.FaceCamera = true
                        beam.LightEmission = 1
                        beam.LightInfluence = 0
                        beam.Parent = beamPart
                        
                        activeLightBeams[p] = beamPart
                    else
                        local beamPart = activeLightBeams[p]
                        beamPart.CFrame = pHrp.CFrame
                        local beam = beamPart:FindFirstChild("MulnesLightBeam")
                        if beam then
                            beam.Color = ColorSequence.new(Color3.fromRGB(espSettings.SightlineColorR, espSettings.SightlineColorG, espSettings.SightlineColorB))
                            beam.Width0 = espSettings.BeamWidth
                            beam.Width1 = espSettings.BeamWidth
                        end
                    end
                else
                    if activeLightBeams[p] then
                        activeLightBeams[p]:Destroy()
                        activeLightBeams[p] = nil
                    end
                end
            end
        end

        -- 16. AIMBOT ENGINE LOOP (FIXED WITH WORKING WALL CHECK & AIMBOT KEYBIND LOGIC)
        local currentTarget = nil
        local shortestDistance = math.huge

        local isKeyHeld = false
        pcall(function()
            if typeof(aimbotSettings.Keybind) == "EnumItem" then
                if aimbotSettings.Keybind.EnumType == Enum.KeyCode then
                    isKeyHeld = UserInputService:IsKeyDown(aimbotSettings.Keybind)
                elseif aimbotSettings.Keybind.EnumType == Enum.UserInputType then
                    isKeyHeld = UserInputService:IsMouseButtonPressed(aimbotSettings.Keybind)
                end
            end
        end)

        if aimbotSettings.Enabled then
            for _, p in pairs(Players:GetPlayers()) do
                if p ~= player then
                    local pChar = p.Character
                    local pHrp = pChar and pChar:FindFirstChild("HumanoidRootPart")
                    local pHum = pChar and pChar:FindFirstChild("Humanoid")
                    
                    local isTeam = false
                    if aimbotSettings.TeamCheck and p.Team and player.Team and p.Team == player.Team then
                        isTeam = true
                    end

                    if pHrp and pHum and pHum.Health > 0 and not isTeam and pChar then
                        local candidateParts = {
                            pChar:FindFirstChild("Head"),
                            pChar:FindFirstChild("UpperTorso") or pChar:FindFirstChild("Torso"),
                            pChar:FindFirstChild("LowerTorso") or pChar:FindFirstChild("HumanoidRootPart")
                        }
                        for _, part in pairs(candidateParts) do
                            if part then
                                local screenPos, onScreen = camera:WorldToViewportPoint(part.Position)
                                if onScreen then
                                    local mousePos = UserInputService:GetMouseLocation()
                                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                                    if dist < shortestDistance then
                                        local passesWallCheck = true
                                        if aimbotSettings.WallCheck then
                                            local origin = camera.CFrame.Position
                                            local direction = (part.Position - origin)
                                            local raycastParams = RaycastParams.new()
                                            raycastParams.FilterType = Enum.RaycastFilterType.Exclude
                                            raycastParams.FilterDescendantsInstances = {player.Character, pChar}
                                            raycastParams.IgnoreWater = true
                                            
                                            local result = Workspace:Raycast(origin, direction, raycastParams)
                                            if result then
                                                local hitPart = result.Instance
                                                if hitPart and hitPart.Transparency < 0.5 and hitPart.CanCollide then
                                                    passesWallCheck = false
                                                end
                                            end
                                        end

                                        if passesWallCheck then
                                            shortestDistance = dist
                                            currentTarget = part
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end

            if isKeyHeld and currentTarget then
                local targetPos = currentTarget.Position
                if aimbotSettings.BulletSpreadReactor then
                    targetPos = targetPos + Vector3.new(0, 0.1, 0)
                end
                
                local currentCamCF = camera.CFrame
                local targetCamCF = CFrame.new(currentCamCF.Position, targetPos)
                local smoothness = math.clamp(aimbotSettings.Smoothness, 1, 50)
                camera.CFrame = currentCamCF:Lerp(targetCamCF, 1 / smoothness)
            end
        end
    end)
end
